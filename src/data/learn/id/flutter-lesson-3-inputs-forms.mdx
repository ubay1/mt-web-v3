# Interaktivitas, Form & Validasi Input

Membangun interaksi sentuhan dan formulir input data yang aman dengan validasi real-time di Flutter.

---

## 1. Sentuhan & Gestur: `GestureDetector` vs `InkWell`

- `GestureDetector`: Menangkap semua jenis gestur (klik, double tap, long press, drag, scale) tanpa efek animasi visual.
- `InkWell`: Menangkap klik sekaligus menghasilkan efek riak gelombang air (*Material ripple effect*). Harus diletakkan di dalam widget `Material` untuk menampilkan riak secara benar.

```dart
// Custom Ripple Card
Material(
  color: Colors.transparent,
  child: InkWell(
    borderRadius: BorderRadius.circular(12),
    onTap: () {
      print('Kartu ditekan dengan efek ripple');
    },
    child: Container(
      padding: const EdgeInsets.all(16),
      child: const Text('Klik Saya!'),
    ),
  ),
)
```

## 2. Formulir & Validasi dengan `TextFormField`

Untuk membuat formulir yang aman dan memiliki validasi bawaan, gunakan widget `Form` yang dikontrol oleh `GlobalKey<FormState>`:

```dart
class LoginForm extends StatefulWidget {
  const LoginForm({super.key});

  @override
  State<LoginForm> createState() => _LoginFormState();
}

class _LoginFormState extends State<LoginForm> {
  final _formKey = GlobalKey<FormState>();
  final _emailController = TextEditingController();
  final _passwordController = TextEditingController();
  bool _obscurePassword = true;

  @override
  void dispose() {
    // Selalu bersihkan TextEditingController untuk mencegah memory leak!
    _emailController.dispose();
    _passwordController.dispose();
    super.dispose();
  }

  void _submitForm() {
    // Memicu seluruh fungsi validator di dalam Form
    if (_formKey.currentState!.validate()) {
      final email = _emailController.text;
      final password = _passwordController.text;
      print('Form Valid! Mengirim email: $email');
    }
  }

  @override
  Widget build(BuildContext context) {
    return Form(
      key: _formKey,
      child: Column(
        children: [
          // Input Email
          TextFormField(
            controller: _emailController,
            keyboardType: TextInputType.emailAddress,
            decoration: const InputDecoration(
              labelText: 'Email Address',
              prefixIcon: Icon(Icons.email_outlined),
              border: OutlineInputBorder(),
            ),
            validator: (value) {
              if (value == null || value.trim().isEmpty) {
                return 'Email tidak boleh kosong';
              }
              if (!value.contains('@')) {
                return 'Format email tidak valid';
              }
              return null; // Valid!
            },
          ),
          const SizedBox(height: 16),

          // Input Password
          TextFormField(
            controller: _passwordController,
            obscureText: _obscurePassword,
            decoration: InputDecoration(
              labelText: 'Password',
              prefixIcon: const Icon(Icons.lock_outline),
              border: const OutlineInputBorder(),
              suffixIcon: IconButton(
                icon: Icon(_obscurePassword ? Icons.visibility : Icons.visibility_off),
                onPressed: () => setState(() => _obscurePassword = !_obscurePassword),
              ),
            ),
            validator: (value) {
              if (value == null || value.length < 6) {
                return 'Password minimal 6 karakter';
              }
              return null;
            },
          ),
          const SizedBox(height: 24),

          // Submit Button
          SizedBox(
            width: double.infinity,
            height: 50,
            child: ElevatedButton(
              onPressed: _submitForm,
              child: const Text('Masuk ke Akun'),
            ),
          ),
        ],
      ),
    );
  }
}
```
