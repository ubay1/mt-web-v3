# Integrasi REST API, JSON Serialization & FutureBuilder

Menghubungkan aplikasi Flutter ke server backend dan mengonversi format JSON menjadi objek model Dart yang *type-safe*.

---

## 1. Membuat Model Data & JSON Serialization

Selalu buat class model untuk memetakan respon JSON agar kode aman dari kesalahan pengetikan nama field (*typo*):

```dart
class Article {
  final int id;
  final String title;
  final String author;

  const Article({
    required this.id,
    required this.title,
    required this.author,
  });

  // Factory constructor untuk mengubah Map JSON menjadi Object Dart
  factory Article.fromJson(Map<String, dynamic> json) {
    return Article(
      id: json['id'] as int,
      title: json['title'] as String,
      author: json['author'] as String? ?? 'Admin',
    );
  }

  // Method untuk mengubah Object Dart kembali menjadi Map JSON
  Map<String, dynamic> toJson() => {
    'id': id,
    'title': title,
    'author': author,
  };
}
```

## 2. Melakukan HTTP Request

Menggunakan package `http` bawaan atau `dio`:

```dart
import 'dart:convert';
import 'package:http/http.dart' as http;

class ArticleService {
  static const String baseUrl = 'https://api.example.com/v1/articles';

  Future<List<Article>> fetchArticles() async {
    final response = await http.get(
      Uri.parse(baseUrl),
      headers: {'Accept': 'application/json'},
    );

    if (response.statusCode == 200) {
      final List<dynamic> jsonList = jsonDecode(response.body);
      return jsonList.map((item) => Article.fromJson(item)).toList();
    } else {
      throw Exception('Gagal memuat artikel. Status: ${response.statusCode}');
    }
  }
}
```

## 3. Menampilkan Data Asinkron dengan `FutureBuilder`

`FutureBuilder` adalah widget yang secara otomatis menangani siklus hidup pengambilan data (loading, error, success):

```dart
class ArticleListPage extends StatefulWidget {
  const ArticleListPage({super.key});

  @override
  State<ArticleListPage> createState() => _ArticleListPageState();
}

class _ArticleListPageState extends State<ArticleListPage> {
  late Future<List<Article>> _articlesFuture;

  @override
  void initState() {
    super.initState();
    // Simpan future di initState agar tidak terpanggil berulang saat render ulang!
    _articlesFuture = ArticleService().fetchArticles();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Daftar Artikel')),
      body: FutureBuilder<List<Article>>(
        future: _articlesFuture,
        builder: (context, snapshot) {
          // 1. State: Sedang Loading
          if (snapshot.connectionState == ConnectionState.waiting) {
            return const Center(child: CircularProgressIndicator());
          }

          // 2. State: Terjadi Error
          if (snapshot.hasError) {
            return Center(
              child: Text('Terjadi error: ${snapshot.error}'),
            );
          }

          // 3. State: Data Berhasil Didapatkan
          final articles = snapshot.data ?? [];
          if (articles.isEmpty) {
            return const Center(child: Text('Tidak ada artikel'));
          }

          return ListView.builder(
            itemCount: articles.length,
            itemBuilder: (context, index) {
              final item = articles[index];
              return ListTile(
                title: Text(item.title),
                subtitle: Text('Penulis: ${item.author}'),
              );
            },
          );
        },
      ),
    );
  }
}
```
