## React 19: Actions, `useActionState` & `useFormStatus`

React 19 introduces **Actions**, `useActionState`, and `useFormStatus` to simplify form handling and asynchronous transitions.

---

### 1a. Actions & `useTransition`

Use `useTransition` to track pending state automatically for async functions without manually setting `useState` loading states.

```jsx
"use client";
import { useState, useTransition } from "react";

function UpdateNameStartTransition() {
  const [name, setName] = useState("");
  const [error, setError] = useState(null);
  const [isPending, startTransition] = useTransition();

  const handleSubmit = () => {
    startTransition(async () => {
      const { error, data } = await ping();
      if (error) {
        setError(error);
        return;
      }
      setName(data);
    });
  };

  return (
    <div>
      <input value={name} onChange={(e) => setName(e.target.value)} />
      <button onClick={handleSubmit} disabled={isPending}>
        {isPending ? "Saving..." : "Save via Transition"}
      </button>
    </div>
  );
}
```
